    // ============================================================
    // CONSTRUCTOR
    // ============================================================
    public FoodClassifier(Context context) {
        try {
            Log.d(TAG, "Loading model from assets...");


            // Check if file exists in assets
            String[] files = context.getAssets().list("");
            boolean found = false;
            for (String f : files) {
                if (f.equals("food_model.tflite")) {
                    found = true;
                    break;
                }
            }
            if (!found) {
                Log.e(TAG, "food_model.tflite not found in assets!");
                Log.d(TAG, "Available files: " + java.util.Arrays.toString(files));
                return;
            }


            // Load the model from assets
            InputStream modelStream = context.getAssets().open("food_model.tflite");
            byte[] modelBytes = new byte[modelStream.available()];
            modelStream.read(modelBytes);
            modelStream.close();

            ByteBuffer modelBuffer = ByteBuffer.allocateDirect(modelBytes.length);
            modelBuffer.put(modelBytes);
            modelBuffer.rewind();

            interpreter = new Interpreter(modelBuffer);
            Log.d(TAG, "Interpreter created successfully");

            // Check input tensor type
            //int inputType = interpreter.getInputTensor(0).dataType();
            DataType inputType = interpreter.getInputTensor(0).dataType();
            //isQuantized = (inputType == 1); // 1 = UINT8
            isQuantized = (inputType == DataType.UINT8);
            Log.d(TAG, "Model loaded. Input type: " + (isQuantized ? "UINT8" : "FLOAT32"));
            Log.d(TAG, "Input shape: " + java.util.Arrays.toString(interpreter.getInputTensor(0).shape()));
        } catch (Exception e) {
           // e.printStackTrace();
            Log.e(TAG, "Failed to load model: " + e.getMessage());
            interpreter = null;
        }
    }

    // ============================================================
    // CLASSIFY METHOD
    // ============================================================
    public String classifyFood(Bitmap bitmap) {
        if (interpreter == null) {
            Log.e(TAG, "Interpreter is null");
            return "food";
        }

        // Get input and output tensor details
        int[] inputShape = interpreter.getInputTensor(0).shape();
        DataType inputType = interpreter.getInputTensor(0).dataType();
        Log.d(TAG, "Input shape: " + java.util.Arrays.toString(inputShape));
        Log.d(TAG, "Input type: " + inputType.toString());

        // Resize image to match model input
        Bitmap resized = Bitmap.createScaledBitmap(bitmap, inputShape[1], inputShape[2], true);

        // Prepare input buffer based on data type
        ByteBuffer inputBuffer;
        if (inputType == DataType.UINT8) {
            // UINT8: 0-255
            inputBuffer = ByteBuffer.allocateDirect(inputShape[1] * inputShape[2] * 3);
            inputBuffer.order(ByteOrder.nativeOrder());
            int[] pixels = new int[inputShape[1] * inputShape[2]];
            resized.getPixels(pixels, 0, inputShape[1], 0, 0, inputShape[1], inputShape[2]);
            for (int pixel : pixels) {
                byte r = (byte) ((pixel >> 16) & 0xFF);
                byte g = (byte) ((pixel >> 8) & 0xFF);
                byte b = (byte) (pixel & 0xFF);
                inputBuffer.put(r);
                inputBuffer.put(g);
                inputBuffer.put(b);
            }
        } else {
            // FLOAT32: 0.0-1.0
            inputBuffer = ByteBuffer.allocateDirect(inputShape[1] * inputShape[2] * 3 * 4);
            inputBuffer.order(ByteOrder.nativeOrder());
            int[] pixels = new int[inputShape[1] * inputShape[2]];
            resized.getPixels(pixels, 0, inputShape[1], 0, 0, inputShape[1], inputShape[2]);
            for (int pixel : pixels) {
                float r = ((pixel >> 16) & 0xFF) / 255.0f;
                float g = ((pixel >> 8) & 0xFF) / 255.0f;
                float b = (pixel & 0xFF) / 255.0f;
                inputBuffer.putFloat(r);
                inputBuffer.putFloat(g);
                inputBuffer.putFloat(b);
            }
        }

        inputBuffer.rewind();

        // Prepare output buffer
        int outputSize = interpreter.getOutputTensor(0).shape()[1];
        float[][] output = new float[1][outputSize];
        Log.d(TAG, "Output size: " + outputSize);

        // Run inference
        try {
            interpreter.run(inputBuffer, output);
            Log.d(TAG, "Inference completed");
        } catch (Exception e) {
            Log.e(TAG, "Inference failed: " + e.getMessage());
            return "food";
        }

        // Find best match
        int bestIndex = 0;
        float bestScore = 0;
        for (int i = 0; i < outputSize; i++) {
            if (output[0][i] > bestScore) {
                bestScore = output[0][i];
                bestIndex = i;
            }
        }

        Log.d(TAG, "Best index: " + bestIndex + ", score: " + bestScore);
        //2608251213
        String modelOutput = FOOD_CLASSES[bestIndex];
        //2608251213

        //try here,change if it doesn't work it can conflict with the below
        // Check if user has a correction for this output
        String correction = dbHelper.getCorrection(modelOutput);
        if (correction != null) {
            return correction;  // Use user's correction
        }

        // Otherwise, use mapping
        if (NAME_MAP.containsKey(modelOutput)) {
            return NAME_MAP.get(modelOutput);
        }
        return modelOutput.replace("_", " ");

        //


        // Return food name if confidence > threshold
        if (bestScore > 0.2 && bestIndex < FOOD_CLASSES.length) {
            //String modelOutput = FOOD_CLASSES[bestIndex];
            modelOutput = FOOD_CLASSES[bestIndex];
            Log.d(TAG, "Recognized: " + modelOutput);
            if (NAME_MAP.containsKey(modelOutput)) {
                return NAME_MAP.get(modelOutput);
            }
            return modelOutput.replace("_", " ");
        }

        return "food";
    }

    // HELPER: Convert Bitmap to ByteBuffer
    // ============================================================
    private ByteBuffer convertBitmapToByteBuffer(Bitmap bitmap) {
        ByteBuffer byteBuffer = ByteBuffer.allocateDirect(INPUT_SIZE * INPUT_SIZE * PIXEL_SIZE * 4);
        byteBuffer.order(ByteOrder.nativeOrder());

        int[] pixels = new int[INPUT_SIZE * INPUT_SIZE];
        bitmap.getPixels(pixels, 0, INPUT_SIZE, 0, 0, INPUT_SIZE, INPUT_SIZE);

        for (int pixel : pixels) {
            float r = ((pixel >> 16) & 0xFF) / 255.0f;
            float g = ((pixel >> 8) & 0xFF) / 255.0f;
            float b = (pixel & 0xFF) / 255.0f;
            byteBuffer.putFloat(r);
            byteBuffer.putFloat(g);
            byteBuffer.putFloat(b);
        }
        return byteBuffer;
    }
